# pg_vectorsearch build options

option('simd',
  type: 'combo',
  choices: ['full', 'compiler', 'none'],
  value: 'full',
  description: '''SIMD implementation mode:
  - full: target_clones baseline + hand-optimized intrinsics (best performance)
  - compiler: target_clones only (compiler-generated vectorization)
  - none: truly scalar (no vectorization, debugging/baseline)'''
)

option('native',
  type: 'boolean',
  value: false,
  description: '''Use -march=native for local builds.
  Enables all CPU-specific instructions for the build machine.
  Produces non-portable binaries optimized for local benchmarking.'''
)

option('blas',
  type: 'combo',
  choices: ['auto', 'enabled', 'disabled'],
  value: 'auto',
  description: '''Use BLAS for matrix operations.
  - auto: detect and use if available (default)
  - enabled: require BLAS, fail if not found
  - disabled: never use BLAS
  Uses cblas_sgemm for matrix multiplication in RaBitQ encoding.
  Install CBLAS library for better performance: OpenBLAS, AOCL-BLIS, Intel MKL.'''
)

option('hdf5',
  type: 'feature',
  value: 'auto',
  description: '''Enable HDF5 dataset loading for benchmarks.
  Auto-detected when libhdf5 is installed.
  Install: libhdf5-dev (Debian/Ubuntu), hdf5 (Arch).'''
)

option('postgresql',
  type: 'feature',
  value: 'auto',
  description: '''Build pg_vectorsearch PostgreSQL extension.
  - auto: detect and build if PostgreSQL is available (default)
  - enabled: require PostgreSQL, fail if not found
  - disabled: never build the extension'''
)

option('pg_config',
  type: 'string',
  value: '',
  description: 'Path to pg_config (auto-detected if not specified)'
)

option('pg_srcdir',
  type: 'string',
  value: '',
  description: 'Path to PostgreSQL source directory (for internal headers)'
)

option('tools',
  type: 'feature',
  value: 'auto',
  description: '''Build the developer tools: the vectorsearch CLI, the
  standalone shared library (Python bindings), and the unit tests.
  - auto: enabled in a git checkout, disabled when building from a release
    tarball (no .git directory) — release-artifact consumers build only the
    PostgreSQL extension unless they opt in
  - enabled: always build them
  - disabled: never build them
  PostgreSQL regression tests are unaffected by this option.'''
)

option('tap_tests',
  type: 'feature',
  value: 'auto',
  description: '''Run TAP tests (multi-cluster scenarios such as streaming
  replication). Needs perl, PostgreSQL::Test from the target PostgreSQL's
  pgxs tree, and IPC::Run.
  - auto: run them when those are present, skip otherwise
  - enabled: require them, and fail configuration if they are missing --
    use this in CI so a missing prerequisite cannot silently drop coverage
  - disabled: never run them'''
)
